\begin{frame}[plain]
    \immuneSectionHeader{2.}{ESTADO DEL ARTE}{Fallos del Deep Learning Estándar y Paradigma Pasivo}

    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.53\textwidth}
            {\bfseries\fontsize{10.5}{12}\selectfont\color{immuneDarkTeal}¿POR QUÉ FRACASAN LOS MODELOS DE AUDIO COMUNES?}
            
            \apartadoImmune{1. Escalado Relativo (Min-Max por Muestra)}
            {\fontsize{7.7}{9.4}\selectfont
            Los modelos estándar normalizan cada audio entre su propio mínimo y máximo. En ventanas de silencio, esto amplifica el siseo electrónico del preamplificador al $100\%$ y dispara \textbf{falsas alarmas}. \textbf{AURA 5} fija la escala a los decibelios físicos reales del sensor ($[-80, 0]\text{ dBFS}$).}

            \apartadoImmune{2. Ignorar que el Audio es una Serie Temporal}
            {\fontsize{7.7}{9.4}\selectfont
            Una CNN pura ve el espectrograma como una foto estática sin memoria del pasado. \textbf{AURA 5} integra capas recurrentes (\textbf{GRU}) que recuerdan los instantes previos para distinguir patrones de golpes intencionales.}

            \apartadoImmune{3. Fuga de Datos (\textit{Data Leakage}) Ambiental}
            {\fontsize{7.7}{9.4}\selectfont
            Mezclar la misma grabación de ruido de fondo en \textit{Train} y \textit{Test} hace que la red memorice el ruido en lugar de aprender la señal de vida.}
        \end{column}
        
        \begin{column}{0.44\textwidth}
            \tarjetaConPestana{5.8cm}{2.5cm}{\textless DESARROLLO PASIVO (SIN ETAPA D/A)\textgreater}{%

                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Seguridad:} No inyecta vibraciones que provoquen derrumbes secundarios.
                    \item \textbf{Autonomía y Sigilo:} Maximiza la batería en campo y mantiene limpio el canal elástico.
                \end{itemize}}%

                $Y(e^{j\Omega}) = X(e^{j\Omega}) H(e^{j\Omega}) H^*(e^{j\Omega}) = X(e^{j\Omega}) |H(e^{j\Omega})|^2$%
            }

            {\fontsize{7.5}{9.2}\selectfont
            \textbf{¿Por qué es vital para las Series Temporales?}
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item Un filtro digital común limpia el ruido pero retrasa distintas frecuencias en distintos tiempos, deformando el ritmo del código Morse.
                \item Al filtrar hacia adelante ($H(e^{j\Omega})$) y luego invertido hacia atrás ($H^*(e^{j\Omega})$), las fases se cancelan ($\phi_{\text{total}}(\Omega) = 0$) y el \textbf{retraso temporal es exactamente cero ($\tau_g(\Omega) = 0$)}.
                \item Así eliminamos la vibración del suelo ($<150\text{ Hz}$) dejando cada impacto en su milisegundo exacto.
            \end{itemize}}

            \tarjetaConPestana{6.7cm}{4.1cm}{\textless REPOSITORIOS SEPARADOS (FASE 1)\textgreater}{%

                {\fontsize{7.6}{9.3}\selectfont
                \textbf{El Riesgo:} Si se utiliza la misma grabación real de una excavadora para contaminar audios sintéticos de entrenamiento y de validación, la red neuronal memoriza la excavadora en lugar de aprender a detectar a la víctima.\par\vspace{5pt}
                \textbf{Solución en AURA 5:} Desde el disco duro en la Fase 1, las grabaciones de ruido real de la Zona Cero se dividen en \textbf{dos repositorios separados}:
                \begin{itemize}
                    \setlength{\itemsep}{2pt}
                    \item \textbf{Pool Train ($80\%$):} Exclusivo para generar y mezclar muestras de entrenamiento.
                    \item \textbf{Pool Validación ($20\%$):} Exclusivo para evaluación; jamás es visto durante el entrenamiento.
                \end{itemize}}%
            }
        \end{column}

        \begin{column}{0.48\textwidth}
            \tarjetaConPestana{6.3cm}{4.1cm}{\textless PARTICIÓN AGRUPADA (FASE 2)\textgreater}{%
                {\bfseries\fontsize{8.5}{10}\selectfont\color{immuneDarkTeal}División por Archivo (\texttt{StratifiedGroupKFold})}\par\vspace{4pt}
                {\fontsize{7.6}{9.3}\selectfont
                \textbf{El Riesgo:} Cuando una grabación larga real se divide en varios fragmentos de $4.0\text{ s}$ (\texttt{audio\_part0}, \texttt{audio\_part1}...), un split aleatorio simple envía trozos de la misma persona tanto a \textit{Train} como a \textit{Test}.\par\vspace{5pt}
                \textbf{Solución en AURA 5:}
                \begin{itemize}
                    \setlength{\itemsep}{2pt}
                    \item El particionador extrae la raíz original del archivo (\texttt{grupo\_base}) y aplica \textbf{\texttt{StratifiedGroupKFold}}.
                    \item Todos los fragmentos de un mismo evento van juntos a Entrenamiento ($52\,417$ tensores) o juntos a Validación ($13\,159$ tensores).
                \end{itemize}}%
            }
        \end{column}
    \end{columns}
